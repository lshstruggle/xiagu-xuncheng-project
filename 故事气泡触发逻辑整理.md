// ============ 1. 启动位置监听 ============
const startLocationWatch = useCallback(() => {
  Taro.startLocationUpdate({
    type: 'gcj02',
    success: () => {
      Taro.onLocationChange((res) => {
        const { latitude, longitude } = res
        setUserLat(latitude)
        setUserLng(longitude)
        checkNearbyPOI(latitude, longitude)
        
        // ⭐ 关键：每次位置变化检查故事触发
        checkStoryTriggerAtLocation(latitude, longitude)
      })
    }
  })
}, [])

// ============ 2. 故事触发检测函数 ============
const checkStoryTriggerAtLocation = (lat: number, lng: number) => {
  // 只在探索模式下检测
  if (exploreMode !== 'explore') return

  // 调用服务层检测
  const result = checkStoryTrigger(lat, lng)
  
  if (result.triggered && result.node) {
    // 避免重复显示同一节点
    if (currentStoryNode?.id !== result.node.id) {
      setCurrentStoryNode(result.node)
      setShowStoryDialog(true)
      
      // 新章节提示
      if (result.isNewChapter && result.chapterTitle) {
        setStoryChapterTitle(result.chapterTitle)
      }
    }
  }
}

## 2️⃣ 服务端触发检测逻辑

### 文件：`src/services/story.ts`

```typescript
/**
 * 检查是否到达故事触发地点
 */
export function checkStoryTrigger(
  userLat: number,
  userLng: number
): StoryTriggerResult {
  const progress = getCurrentStoryProgress()
  if (!progress || progress.status !== 'ongoing') {
    return { triggered: false }
  }

  const story = getStoryById(progress.storyId)
  if (!story) return { triggered: false }

  // 获取当前节点
  const currentNode = story.nodes[progress.currentNodeId]
  if (!currentNode || !currentNode.location) {
    return { triggered: false }
  }

  // 计算距离
  const { location } = currentNode
  const distance = calculateDistance(
    userLat,
    userLng,
    location.lat,
    location.lng
  )

  // 触发范围（默认200米）
  const triggerRadius = location.radius || 200

  // 在范围内则触发
  if (distance <= triggerRadius) {
    const isNewChapter =
      currentNode.chapter !== undefined &&
      currentNode.id === story.chapters.find(c => c.id === currentNode.chapter)?.nodes[0]

    const chapter = story.chapters.find(c => c.id === currentNode.chapter)

    return {
      triggered: true,
      node: currentNode,
      isNewChapter,
      chapterTitle: chapter?.title
    }
  }

  return { triggered: false }
}

/**
 * 计算两点距离（米）
 */
function calculateDistance(
  lat1: number,
  lng1: number,
  lat2: number,
  lng2: number
): number {
  const R = 6371000 // 地球半径（米）
  const dLat = (lat2 - lat1) * Math.PI / 180
  const dLng = (lng2 - lng1) * Math.PI / 180
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos(lat1 * Math.PI / 180) *
    Math.cos(lat2 * Math.PI / 180) *
    Math.sin(dLng / 2) *
    Math.sin(dLng / 2)
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
  return R * c
}
```

## 3️⃣ 故事对话框组件

### 文件：`src/components/story-dialog/index.tsx`

```typescript
interface StoryDialogProps {
  node: StoryNode           // 当前故事节点
  onNext: () => void        // 点击下一步
  onChoice: (choiceId: string) => void  // 选择选项
  onSkip: () => void        // 跳过/关闭
  autoPlay?: boolean        // 自动播放TTS
}

export default function StoryDialog({
  node,
  onNext,
  onChoice,
  onSkip,
  autoPlay = true
}: StoryDialogProps) {
  const [displayText, setDisplayText] = useState('')
  const [isTyping, setIsTyping] = useState(false)
  const [showChoices, setShowChoices] = useState(false)

  // 逐字显示效果
  useEffect(() => {
    if (node.dialog?.content) {
      setDisplayText('')
      setIsTyping(true)
      setShowChoices(false)

      const text = node.dialog.content
      let index = 0

      const typeInterval = setInterval(() => {
        if (index < text.length) {
          setDisplayText(text.slice(0, index + 1))
          index++
        } else {
          setIsTyping(false)
          clearInterval(typeInterval)
          // 显示选项（如果是选择节点）
          if (node.type === 'choice' && node.choices) {
            setShowChoices(true)
          }
        }
      }, 50) // 每50ms显示一个字

      return () => clearInterval(typeInterval)
    }
  }, [node])

  // 点击屏幕继续
  const handleTap = () => {
    if (isTyping) {
      // 正在打字，显示完整文本
      setDisplayText(node.dialog?.content || '')
      setIsTyping(false)
      if (node.type === 'choice' && node.choices) {
        setShowChoices(true)
      }
    } else if (node.type !== 'choice' && node.type !== 'ending') {
      // 普通对话，进入下一节点
      onNext()
    }
  }

  // 选择选项
  const handleChoice = (choice: StoryChoice) => {
    setShowChoices(false)
    onChoice(choice.id)
  }

  // 渲染...
}
```

## 4️⃣ 节点推进逻辑

### 文件：`src/pages/checkin/index.tsx`

```typescript
// ============ 处理故事对话下一步 ============
const handleStoryNext = () => {
  if (!storyProgress || !currentStoryNode) return

  // 发放奖励（如果有）
  const rewards = claimNodeRewards(storyProgress)
  if (rewards.bondPoints > 0) {
    console.log('获得奖励:', rewards)
  }

  // 推进故事
  const updatedProgress = advanceStory(storyProgress)
  setStoryProgress(updatedProgress)

  // 获取下一个节点
  const story = getStoryById(updatedProgress.storyId)
  if (story) {
    const nextNode = story.nodes[updatedProgress.currentNodeId]
    if (nextNode) {
      setCurrentStoryNode(nextNode)
      // 如果是结局节点，完成故事
      if (nextNode.type === 'ending' && nextNode.ending) {
        completeStory(nextNode.ending.type)
      }
    } else {
      // 故事结束
      setShowStoryDialog(false)
      setCurrentStoryNode(null)
    }
  }
}

// ============ 处理选择 ============
const handleStoryChoice = (choiceId: string) => {
  if (!storyProgress) return

  const updatedProgress = advanceStory(storyProgress, choiceId)
  setStoryProgress(updatedProgress)

  const story = getStoryById(updatedProgress.storyId)
  if (story) {
    const nextNode = story.nodes[updatedProgress.currentNodeId]
    setCurrentStoryNode(nextNode || null)
  }
}

// ============ 跳过/关闭 ============
const handleStorySkip = () => {
  setShowStoryDialog(false)
}
```

interface StoryNode {
  id: string
  type: 'dialog' | 'choice' | 'checkin' | 'ending' | 'transition'
  chapter?: string
  location?: {
    name: string
    address: string
    lat: number
    lng: number
    radius?: number  // 触发范围（米）
  }
  dialog?: {
    speaker: string
    speakerAvatar?: string
    content: string
    emotion?: 'normal' | 'happy' | 'sad' | 'excited' | 'thoughtful'
    ttsAudio?: string
  }
  choices?: StoryChoice[]  // 选择分支
  nextNodeId?: string      // 自动跳转的下一个节点
  checkinReward?: {
    bondPoints: number
    fragments: string[]
    poetryLines: string[]
  }
  ending?: {
    type: 'perfect' | 'good' | 'normal' | 'incomplete'
    title: string
    content: string
    rewards: { ... }
  }
}

## 🎯 触发条件总结

| 条件 | 说明 |
|------|------|
| **模式** | 必须为 `explore` 探索模式 |
| **进度** | 必须有进行中的故事进度 |
| **位置** | 必须在当前节点的 `location` 范围内 |
| **距离** | 默认200米内触发（可配置 `radius`）|
| **重复** | 同一节点不会重复触发 |
